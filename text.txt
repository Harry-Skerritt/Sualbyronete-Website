<script define:vars={{ dbPuppies }}>
    // Track State Values
    let currentBreedFilter = "state-one";

    function renderPuppies() {
        const grid = document.getElementById('puppies-grid');
        if (!grid) return;

        // Fetch inputs from our native, secure HTML elements cleanly!
        const genderValue = document.getElementById('gender-select').value;
        const colourValue = document.getElementById('colour-select').value;
        const availabilityValue = document.getElementById('availability-select').value;

        const filteredPuppies = dbPuppies.filter(pup => {
            // Check string properties against database attributes
            const matchesBreed = pup.breed === currentBreedFilter;

            // Allow values to match lowercased entries in database schemas gracefully
            const matchesGender = genderValue === 'all' || pup.gender.toLowerCase() === genderValue;
            const matchesColour = colourValue === 'all' || pup.colour.toLowerCase().replace(/[^a-z]/g, '') === colourValue.replace(/[^a-z]/g, '');
            const matchesStatus = availabilityValue === 'all' || pup.state.toLowerCase() === availabilityValue;

            return matchesBreed && matchesGender && matchesColour && matchesStatus;
        });

        if (filteredPuppies.length === 0) {
            grid.innerHTML = `
                <div class="no-results" style="grid-column: 1/-1; text-align: center; padding: 4rem; color: var(--theme-text); font-family: var(--font-primary, serif); font-size: 1.4rem; font-weight: bold;">
                    No puppies match your selected filter criteria.
                </div>
            `;
            return;
        }

        grid.innerHTML = filteredPuppies.map(pup => `
            <div class="puppy-card">
                <h2>${pup.name}</h2>
                <p><strong>Gender:</strong> ${pup.gender}</p>
                <p><strong>Colour:</strong> ${pup.colour}</p>
                <span class="status-tag">${pup.state}</span>
            </div>
        `).join('');
    }

    function initPageLogic() {
        renderPuppies();

        // Target changes via the explicit native element array hook
        const filterSelects = document.querySelectorAll('[data-filter-select]');
        filterSelects.forEach(select => {
            select.addEventListener('change', renderPuppies);
        });

        window.addEventListener('breedChange', (e) => {
            currentBreedFilter = e.detail.activeBreed;
            renderPuppies();
        });
    }

    initPageLogic();
    document.addEventListener('astro:page-load', initPageLogic);
</script>
